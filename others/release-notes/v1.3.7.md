## 📖 使用方法

### 基本的な使用

```python
from dictsqlite import DictSQLite

# データベースの作成
db = DictSQLite('improved.db')

# 改善されたパフォーマンスでの操作
db['user_data'] = {
    'name': 'Alice',
    'preferences': ['python', 'sqlite', 'json'],
    'settings': {'theme': 'dark', 'notifications': True}
}

# 高速な検索とアクセス

print(f"User: {user['name']}")
```

### 強化された機能

```python
# より安定した並行アクセス
import threading

def worker(db_path, worker_id):
    db = DictSQLite(db_path)
    db[f'worker_{worker_id}'] = f'Data from worker {worker_id}'
    db.close()

# 複数スレッドでの安全な操作
threads = []
for i in range(5):
    t = threading.Thread(target=worker, args=('shared.db', i))
    threads.append(t)
    t.start()

for t in threads:
    t.join()
```

### ベンチマーク

```python
# パフォーマンステスト例
import time
from dictsqlite import DictSQLite

db = DictSQLite('benchmark.db')

# 大量データの書き込みテスト
start_time = time.time()
for i in range(10000):
    db[f'key_{i}'] = {'value': i, 'data': f'test_data_{i}'}

write_time = time.time() - start_time
print(f"10000件の書き込み時間: {write_time:.2f}秒")

# 読み込みテスト
start_time = time.time()
for i in range(0, 10000, 100):
    _ = db[f'key_{i}']

read_time = time.time() - start_time
print(f"100件の読み込み時間: {read_time:.2f}秒")
```
